## 4. STEP-BY-STEP COMPUTATION

### Step 1: Generate ω₃ Weyl Orbit

```python
import numpy as np
from collections import deque

# D6 simple roots
def get_d6_simple_roots():
    roots = []
    for i in range(5):
        r = np.zeros(6)
        r[i] = 1
        r[i+1] = -1
        roots.append(r)
    # α₆ = e₅ + e₆
    r6 = np.zeros(6)
    r6[4] = 1
    r6[5] = 1
    roots.append(r6)
    return roots

# Weyl reflection
def reflect(v, alpha):
    return v - (np.dot(v, alpha)) * alpha

# Generate orbit via BFS
def generate_weyl_orbit(seed, simple_roots, tol=1e-10):
    orbit = [seed]
    seen = {tuple(np.round(seed / tol).astype(int))}
    queue = deque([seed])
    
    while queue:
        v = queue.popleft()
        for alpha in simple_roots:
            w = reflect(v, alpha)
            key = tuple(np.round(w / tol).astype(int))
            if key not in seen:
                seen.add(key)
                orbit.append(w)
                queue.append(w)
    
    return np.array(orbit)

# Generate ω₃ orbit
omega3 = np.array([1, 1, 1, 0, 0, 0], dtype=float)
simple_roots = get_d6_simple_roots()
weights = generate_weyl_orbit(omega3, simple_roots)
print(f"ω₃ orbit size: {len(weights)}")  # Should be 160
```

```python
def build_adjacency_sparse(vectors, threshold=2.0, tol=1e-5):
    n = len(vectors)
    adj = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            d2 = np.sum((vectors[i] - vectors[j])**2)
            if np.abs(d2 - threshold) < tol:
                adj[i, j] = 1
                adj[j, i] = 1
    return adj
```

### Step 4: Build Product-Weighted Laplacian

```python
def build_product_laplacian(adj, xi):
    n = len(xi)
    L = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if adj[i, j] > 0:
                w = xi[i] * xi[j]
                L[i, j] = -w
                L[i, i] += w
    return L

L = build_product_laplacian(adj, xi)
```
